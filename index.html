/* توزيع الدروس */ function distribute(total, days) { const base = Math.floor(total / days); const extra = total % days; return Array.from( { length: days }, (_, index) => { return base + (index < extra ? 1 : 0); } ); } /* إنشاء المهام */ function getTasks() { // الجدول الأصلي if (!plan.rescheduled || !plan.newDistribution) { const distribution = distribute(plan.total, plan.days); let number = 1; return distribution.map(count => { const tasks = []; for (let i = 0; i < count; i++) { tasks.push(number); number++; } return tasks; }); } // بعد إعادة الجدولة const remainingLessons = []; for ( let number = 1; number <= plan.total; number++ ) { const key = "lesson-" + number; if (!plan.completed.includes(key)) { remainingLessons.push(number); } } const result = []; // الأيام السابقة لليوم الحالي for ( let i = 0; i < plan.currentDay - 1; i++ ) { result.push([]); } // توزيع الدروس المتبقية let position = 0; plan.newDistribution.forEach(count => { const tasks = []; for ( let i = 0; i < count && position < remainingLessons.length; i++ ) { tasks.push( remainingLessons[position] ); position++; } result.push(tasks); }); return result; } /* إنجاز المهمة */ function toggleTask(key) { const index = plan.completed.indexOf(key); if (index >= 0) { plan.completed.splice(index, 1); } else { plan.completed.push(key); } savePlan(); renderPlan(); } /* إعادة الجدولة */ function reschedule() { const completed = plan.completed.length; const remaining = Math.max( 0, plan.total - completed ); const daysLeft = Math.max( 1, plan.days - plan.currentDay + 1 ); plan.newDistribution = distribute( remaining, daysLeft ); plan.rescheduled = true; savePlan(); renderPlan(); }
